
CREATE DATABASE healthcareSystem;
USE healthcare_system;



CREATE TABLE patient (
    patient_id INT PRIMARY KEY AUTO_INCREMENT,
    first_name VARCHAR(50) NOT NULL,
    last_name VARCHAR(50) NOT NULL,
    insurance_no VARCHAR(50),
    insurance_provider VARCHAR(100),
    address VARCHAR(255),
    email VARCHAR(100),
    phone VARCHAR(15),
    blood_group VARCHAR(10),
    gender VARCHAR(10),
    dob DATE
);



CREATE TABLE medical_history (
    history_id INT PRIMARY KEY AUTO_INCREMENT,
    patient_id INT NOT NULL,
    history_date DATE,
    allergies VARCHAR(255),
    diagnoses VARCHAR(255),
    medications VARCHAR(255),
    previous_treatment VARCHAR(255),
    remarks VARCHAR(500),

    FOREIGN KEY (patient_id)
        REFERENCES patient(patient_id)
);



CREATE TABLE doctor (
    doctor_id INT PRIMARY KEY AUTO_INCREMENT,
    doctor_name VARCHAR(100) NOT NULL,
    specialization VARCHAR(100),
    degree VARCHAR(100),
    phone VARCHAR(15),
    experience INT,
    joining_date DATE
);



CREATE TABLE encounter (
    encounter_id INT PRIMARY KEY AUTO_INCREMENT,
    patient_id INT NOT NULL,
    doctor_id INT NOT NULL,
    reason VARCHAR(255),
    encounter_type VARCHAR(50),
    `date` DATE,
    discharge_date DATE,

    FOREIGN KEY (patient_id)
        REFERENCES patient(patient_id),

    FOREIGN KEY (doctor_id)
        REFERENCES doctor(doctor_id)
);



CREATE TABLE report (
    report_id INT PRIMARY KEY AUTO_INCREMENT,
    encounter_id INT NOT NULL,
    `date` DATE,
    result VARCHAR(500),

    FOREIGN KEY (encounter_id)
        REFERENCES encounter(encounter_id)
);



CREATE TABLE test (
    test_id INT PRIMARY KEY AUTO_INCREMENT,
    report_id INT NOT NULL,
    prescription_id INT,
    test_type VARCHAR(100),
    test_name VARCHAR(100),
    charges DECIMAL(10,2),
    estimate_time VARCHAR(50),

    FOREIGN KEY (report_id)
        REFERENCES report(report_id)
);



CREATE TABLE medicine (
    medicine_id INT PRIMARY KEY AUTO_INCREMENT,
    encounter_id INT,
    medicine_name VARCHAR(100) NOT NULL,
    medicine_type VARCHAR(50),
    instructions VARCHAR(500),

    FOREIGN KEY (encounter_id)
        REFERENCES encounter(encounter_id)
);



CREATE TABLE prescription (
    prescription_id INT PRIMARY KEY AUTO_INCREMENT,
    medicine_id INT NOT NULL,
    doctor_id INT NOT NULL,
    dispensing VARCHAR(100),
    duration VARCHAR(50),
    medications VARCHAR(255),
    dosage VARCHAR(100),
    prescription_date DATE,

    FOREIGN KEY (medicine_id)
        REFERENCES medicine(medicine_id),

    FOREIGN KEY (doctor_id)
        REFERENCES doctor(doctor_id)
);


ALTER TABLE test
ADD CONSTRAINT fk_test_prescription
FOREIGN KEY (prescription_id)
REFERENCES prescription(prescription_id);



CREATE TABLE surgery_note (
    surgery_noteid INT PRIMARY KEY AUTO_INCREMENT,
    surgery_id INT,
    encounter_id INT NOT NULL,
    note_date DATE,
    observations VARCHAR(500),
    recommendations VARCHAR(500),

    FOREIGN KEY (encounter_id)
        REFERENCES encounter(encounter_id)
);



CREATE TABLE surgery (
    surgery_id INT PRIMARY KEY AUTO_INCREMENT,
    patient_id INT NOT NULL,
    surgery_noteid INT,
    surgery_name VARCHAR(150),
    surgery_date DATE,
    charges DECIMAL(10,2),
    status VARCHAR(50),

    FOREIGN KEY (patient_id)
        REFERENCES patient(patient_id),

    FOREIGN KEY (surgery_noteid)
        REFERENCES surgery_note(surgery_noteid)
);

ALTER TABLE surgery_note
ADD CONSTRAINT fk_surgery_note_surgery
FOREIGN KEY (surgery_id)
REFERENCES surgery(surgery_id);

SELECT
    p.patient_id,
    p.first_name,
    p.last_name,
    d.doctor_name,
    d.specialization,
    e.reason,
    e.encounter_type,
    e.date
FROM patient p
INNER JOIN encounter e
    ON p.patient_id = e.patient_id
INNER JOIN doctor d
    ON e.doctor_id = d.doctor_id;


SELECT
    p.first_name,
    p.last_name,
    mh.diagnoses,
    mh.allergies,
    mh.medications,
    mh.remarks
FROM patient p
INNER JOIN medical_history mh
    ON p.patient_id = mh.patient_id;


SELECT
    d.doctor_name,
    d.specialization,
    pr.medications,
    pr.dosage,
    pr.duration,
    pr.prescription_date
FROM doctor d
INNER JOIN prescription pr
    ON d.doctor_id = pr.doctor_id;


SELECT
    p.first_name,
    p.last_name,
    r.result,
    t.test_name,
    t.test_type,
    t.charges
FROM patient p
INNER JOIN encounter e
    ON p.patient_id = e.patient_id
INNER JOIN report r
    ON e.encounter_id = r.encounter_id
INNER JOIN test t
    ON r.report_id = t.report_id;


SELECT
    specialization,
    COUNT(*) AS total_doctors
FROM doctor
GROUP BY specialization;


SELECT
    encounter_type,
    COUNT(*) AS total_encounters
FROM encounter
GROUP BY encounter_type;


SELECT
    status,
    COUNT(*) AS total_surgeries
FROM surgery
GROUP BY status;


SELECT
    test_type,
    SUM(charges) AS total_charges,
    AVG(charges) AS average_charges
FROM test
GROUP BY test_type;


SELECT
    encounter_type,
    COUNT(*) AS total_encounters
FROM encounter
GROUP BY encounter_type
HAVING COUNT(*) > 1;


SELECT
    test_type,
    AVG(charges) AS average_charges
FROM test
GROUP BY test_type
HAVING AVG(charges) > 1000;


SELECT
    status,
    COUNT(*) AS total_surgeries
FROM surgery
GROUP BY status
HAVING COUNT(*) > 2;


SELECT
    test_name,
    test_type,
    charges
FROM test
WHERE charges > (
    SELECT AVG(charges)
    FROM test
);


SELECT
    patient_id,
    first_name,
    last_name
FROM patient
WHERE patient_id IN (
    SELECT patient_id
    FROM encounter
    GROUP BY patient_id
    HAVING COUNT(*) > 1
);


SELECT
    surgery_name,
    charges,
    status
FROM surgery
WHERE charges = (
    SELECT MAX(charges)
    FROM surgery
);


SELECT
    doctor_name,
    specialization,
    experience
FROM doctor
WHERE experience = (
    SELECT MAX(experience)
    FROM doctor
);


DELIMITER //

CREATE PROCEDURE GetAllPatients()
BEGIN
    SELECT *
    FROM patient;
END //

DELIMITER ;

CALL GetAllPatients();


DELIMITER //

CREATE PROCEDURE GetPatientById(IN p_id INT)
BEGIN
    SELECT *
    FROM patient
    WHERE patient_id = p_id;
END //

DELIMITER ;

CALL GetPatientById(1);


DELIMITER //

CREATE PROCEDURE GetDoctorEncounters(IN d_id INT)
BEGIN
    SELECT
        d.doctor_name,
        p.first_name,
        p.last_name,
        e.reason,
        e.encounter_type,
        e.date
    FROM encounter e
    INNER JOIN doctor d
        ON e.doctor_id = d.doctor_id
    INNER JOIN patient p
        ON e.patient_id = p.patient_id
    WHERE d.doctor_id = d_id;
END //

DELIMITER ;

CALL GetDoctorEncounters(1);

CREATE TRIGGER ValidateSurgeryCharges
BEFORE INSERT ON surgery
FOR EACH ROW
BEGIN
    IF NEW.charges < 0 THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Surgery charges cannot be negative';
    END IF;
END //

CREATE TABLE surgery_audit (
    audit_id INT PRIMARY KEY AUTO_INCREMENT,
    surgery_id INT,
    old_status VARCHAR(50),
    new_status VARCHAR(50),
    changed_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

DELIMITER //

CREATE TRIGGER TrackSurgeryStatus
AFTER UPDATE ON surgery
FOR EACH ROW
BEGIN
    IF NOT (OLD.status <=> NEW.status) THEN
        INSERT INTO surgery_audit
            (surgery_id, old_status, new_status)
        VALUES
            (OLD.surgery_id, OLD.status, NEW.status);
    END IF;
END //

DELIMITER ;

SET status = 'Completed'
WHERE surgery_id = 7;
